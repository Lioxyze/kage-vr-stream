function(kage_add_test name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE kage::core)
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4 /utf-8)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endfunction()
