function(kage_add_test name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE kage::core)
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4 /utf-8)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endfunction()

kage_add_test(test_protocol)
kage_add_test(test_video_fragment)
kage_add_test(test_sync_telemetry)
kage_add_test(test_loopback)
set_tests_properties(test_loopback PROPERTIES LABELS integration TIMEOUT 30)
kage_add_test(test_crypto)
